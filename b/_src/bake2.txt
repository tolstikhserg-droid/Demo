def font(sz): return ImageFont.truetype(FONT,sz)
def block(img,lines,x,y,mw,acc,base=150):
    d=ImageDraw.Draw(img); cy=y
    for text,style,sc in lines:
        size=int(base*sc*S); f=font(size)
        while f.getlength(text)>mw and size>60: size-=6; f=font(size)
        tw=f.getlength(text); bb=f.getbbox(text)
        if style=='box':
            pad=int(size*0.14)
            box=Image.new('RGBA',(int(tw+pad*2),int(bb[3]-bb[1]+pad*2)),acc+(255,)).rotate(-2,expand=True,resample=Image.BICUBIC)
            sh=Image.new('RGBA',box.size,(0,0,0,0)); sh.putalpha(box.split()[3].point(lambda v:int(v*0.55)))
            img.alpha_composite(sh.filter(ImageFilter.GaussianBlur(10)),(int(x-pad+14),int(cy+bb[1]-pad+18)))
            img.alpha_composite(box,(int(x-pad),int(cy+bb[1]-pad)))
            d.text((x,cy),text,font=f,fill=(15,15,15))
        else:
            col=(255,255,255) if style=='w' else acc
            sh=Image.new('RGBA',img.size,(0,0,0,0))
            ImageDraw.Draw(sh).text((x+12,cy+16),text,font=f,fill=(0,0,0,210),stroke_width=int(size*0.07),stroke_fill=(0,0,0,210))
            img.alpha_composite(sh.filter(ImageFilter.GaussianBlur(12)))
            d.text((x,cy),text,font=f,fill=col,stroke_width=max(8,int(size*0.055)),stroke_fill=(10,10,10))
        cy+=(bb[3]-bb[1])+int(size*0.28)
def badge(img,t,x,y,bg,fg,size,rot):
    size*=S; f=font(size); tw=f.getlength(t); bb=f.getbbox(t)
    pw,ph=int(tw+size*0.9),int(bb[3]-bb[1]+size*0.75)
    b=Image.new('RGBA',(pw,ph),(0,0,0,0)); dd=ImageDraw.Draw(b)
    dd.rounded_rectangle((0,0,pw-1,ph-1),radius=int(ph*0.25),fill=bg+(255,))
    dd.text((size*0.45,(ph-(bb[3]-bb[1]))/2-bb[1]),t,font=f,fill=fg)
    b=b.rotate(rot,expand=True,resample=Image.BICUBIC)
    sh=Image.new('RGBA',b.size,(0,0,0,0)); sh.putalpha(b.split()[3].point(lambda v:int(v*0.5)))
    img.alpha_composite(sh.filter(ImageFilter.GaussianBlur(14)),(int(x+12),int(y+20)))
    img.alpha_composite(b,(int(x),int(y)))
def shade(img,strength=0.55,width=0.55):
    xx=np.linspace(0,1,W)[None,:]; g=np.clip(1-xx/width,0,1)
    g=(g**1.4*strength*255).astype(np.uint8).repeat(H,0)
    lay=Image.new('RGBA',(W,H),(0,0,0,255)); lay.putalpha(Image.fromarray(g)); img.alpha_composite(lay)
def person(img,path,hfrac=0.94,right=40):
    p=Image.open(path).convert('RGBA'); s=H*hfrac/p.height; p=p.resize((round(p.width*s),round(p.height*s)),Image.LANCZOS)
    if p.width>W*0.6: s2=W*0.6/p.width; p=p.resize((round(p.width*s2),round(p.height*s2)),Image.LANCZOS)
    x=W-p.width-right*S; y=H-p.height
    a=p.split()[3]
    glow=a.filter(ImageFilter.MaxFilter(15)).filter(ImageFilter.GaussianBlur(6))
    g=Image.new('RGBA',p.size,(255,255,255,0)); g.putalpha(glow.point(lambda v:int(v*0.9)))
    sh=Image.new('RGBA',p.size,(0,0,0,0)); sh.putalpha(a.filter(ImageFilter.GaussianBlur(30)).point(lambda v:int(v*0.7)))
    img.alpha_composite(sh,(x-30,y+10)); img.alpha_composite(g,(x,y)); img.alpha_composite(p,(x,y))
slug,src,out=sys.argv[1:4]; c=CFG[slug]; cut=sys.argv[4] if len(sys.argv)>4 else None
im=Image.open(src).convert('RGB')
s=max(W/im.width,H/im.height); im=im.resize((round(im.width*s),round(im.height*s)),Image.LANCZOS)
x0=(im.width-W)//2; y0=(im.height-H)//2; im=im.crop((x0,y0,x0+W,y0+H)).convert('RGBA')
shade(im)
if cut: person(im,cut)
bx,by,bw=c['box']; block(im,c['lines'],bx*S,by*S,bw*S,hexrgb(c['acc']))
for b in c['badges']: badge(im,b['t'],b['x']*S,b['y']*S,hexrgb(b.get('bg','#e61e28')),hexrgb(b.get('fg','#ffffff')),b.get('size',46),b.get('rot',-4))
im=im.convert('RGB'); im.save(out,quality=90)
im.resize((1280,720),Image.LANCZOS).save(out.replace('.jpg','-720.jpg'),quality=88)
